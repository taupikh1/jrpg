async function tapE() { await p.keyboard.down('KeyE'); await p.waitForTimeout(150); await p.keyboard.up('KeyE'); }
async function walkTo(tx, ty, tol = 4, max = 12000) {
  const t0 = Date.now(); const held = new Set();
  const set = async (keys) => {
    for (const k of [...held]) if (!keys.has(k)) { await p.keyboard.up(k); held.delete(k); }
    for (const k of keys) if (!held.has(k)) { await p.keyboard.down(k); held.add(k); }
  };
  while (Date.now() - t0 < max) {
    const s = await st();
    if (s.mode === 'dialogue') { await set(new Set()); return 'dialogue-opened'; }
    const dx = tx - s.x, dy = ty - s.y; const k = new Set();
    if (Math.abs(dx) > tol) k.add(dx > 0 ? 'ArrowRight' : 'ArrowLeft');
    if (Math.abs(dy) > tol) k.add(dy > 0 ? 'ArrowDown' : 'ArrowUp');
    if (!k.size) { await set(new Set()); return 'arrived'; }
    await set(k); await p.waitForTimeout(40);
  }
  await set(new Set());
  return 'timeout ' + JSON.stringify(await st());
}
