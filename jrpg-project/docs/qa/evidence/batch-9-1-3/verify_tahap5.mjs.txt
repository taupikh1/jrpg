import { chromium } from '/opt/npm-tools/node_modules/playwright/index.mjs';
const OUT = 'docs/qa/evidence/batch-9-1-3';
const ENTER_ZONE = ({ zoneId, spawnId }) => {
  const s = window.__JRPG__.scenes.current; const pid = s.player.id;
  s.zones.transitionTo(zoneId, spawnId);
  const zone = s.zones.activeZone;
  for (const e of [...s.entities.all()]) if (e.id !== pid) s.entities.remove(e.id);
  s.player.zoneId = zone.id;
  const sp = zone.spawnPoints.find((x) => x.id === spawnId);
  s.player.x = sp.x; s.player.y = sp.y;
  const mov = s.player.get('movement'); if (mov) mov.collision = s.zones.collision;
  s._spawnZoneNpcs(zone); s._applyCamera(); s._rebuildLayers();
  return zone.id;
};
const b = await chromium.launch({ args: ['--no-sandbox'] });
const p = await (await b.newContext({ viewport: { width: 960, height: 540 } })).newPage();
const errors = []; p.on('pageerror', (e) => errors.push(e.message)); p.on('console', (m) => { if (m.type() === 'error') errors.push(m.text()); });
await p.goto('http://localhost:8137/index.html'); await p.waitForTimeout(1500);
for (let i = 0; i < 60; i++) { if ((await p.evaluate(() => window.__JRPG__.scenes.current.mode)) !== 'dialogue') break; await p.keyboard.press('Enter'); await p.waitForTimeout(120); }
const zone = () => p.evaluate(() => window.__JRPG__.scenes.current.zones.activeZone?.id);
const toast = () => p.evaluate(() => window.__JRPG__.scenes.current._toast);
const pos = () => p.evaluate(() => { const s = window.__JRPG__.scenes.current; return { x: Math.round(s.player.x), y: Math.round(s.player.y) }; });

// Locked: walk into the road trigger from the forest with no chapter flags
await p.evaluate(ENTER_ZONE, { zoneId: 'forest', spawnId: 'from_village' });
await p.waitForTimeout(200);
await p.evaluate(() => { const s = window.__JRPG__.scenes.current; s.player.x = 200; s.player.y = 190; });
await p.waitForTimeout(300);
await p.keyboard.down('ArrowDown'); await p.waitForTimeout(700); await p.keyboard.up('ArrowDown');
await p.waitForTimeout(200);
console.log('LOCKED zone:', await zone(), 'pos', JSON.stringify(await pos()), 'toast:', await toast());
await p.screenshot({ path: `${OUT}/d3_locked_road.png` });

// Unlocked: after chapter 2 completes, the same walk enters the road
await p.evaluate(() => { const s = window.__JRPG__.scenes.current; s._ch2Flags = Object.assign(Object.create(null), s._ch2Flags || {}, { ch2_completed: true }); s.player.x = 200; s.player.y = 190; });
await p.waitForTimeout(300);
await p.keyboard.down('ArrowDown'); await p.waitForTimeout(700); await p.keyboard.up('ArrowDown');
await p.waitForTimeout(400);
console.log('UNLOCKED zone:', await zone(), 'toast:', await toast());
console.log('errors', errors);
await b.close();
