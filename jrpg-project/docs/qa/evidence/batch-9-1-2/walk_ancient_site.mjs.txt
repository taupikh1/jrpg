import { chromium } from '/opt/npm-tools/node_modules/playwright/index.mjs';
const b = await chromium.launch({ args:['--no-sandbox'] });
const ctx = await b.newContext({ viewport:{width:960,height:540} });
const p = await ctx.newPage();
const logs=[]; p.on('console',m=>{if(m.type()==='error')logs.push(m.text())}); p.on('pageerror',e=>logs.push('PAGEERROR '+e.message));
await p.goto('http://localhost:8137/index.html'); await p.waitForTimeout(1500);
const st=()=>p.evaluate(()=>{const s=window.__JRPG__.scenes.current;return {mode:s.mode,zone:s.zones?.activeZone?.id,x:Math.round(s.player.x),y:Math.round(s.player.y)}});
// advance opening dialogue with real key presses
for(let i=0;i<60;i++){ const s=await st(); if(s.mode!=='dialogue')break; await p.keyboard.press('Enter'); await p.waitForTimeout(120); }
console.log('start',await st());
async function goto(tx,ty,tol=6,max=15000){
  const t0=Date.now(); let held=new Set();
  const set=async(keys)=>{for(const k of [...held]) if(!keys.has(k)){await p.keyboard.up(k);held.delete(k)} for(const k of keys) if(!held.has(k)){await p.keyboard.down(k);held.add(k)}};
  const z0=(await st()).zone;
  while(Date.now()-t0<max){ const s=await st(); if(s.zone!==z0){await set(new Set());return 'zone:'+s.zone}
    if(s.mode==='dialogue'){await set(new Set());await p.keyboard.press('Enter');await p.waitForTimeout(150);continue}
    const dx=tx-s.x, dy=ty-s.y; const k=new Set();
    if(Math.abs(dx)>tol)k.add(dx>0?'ArrowRight':'ArrowLeft'); if(Math.abs(dy)>tol)k.add(dy>0?'ArrowDown':'ArrowUp');
    if(!k.size){await set(k);return 'arrived'} await set(k); await p.waitForTimeout(40);}
  await set(new Set()); return 'timeout '+JSON.stringify(await st());
}
console.log('to forest', await goto(160,222), await st());
console.log('to road', await goto(192,222), await st());
console.log('along road', await goto(311,96,3), await st());
await p.screenshot({path:'docs/qa/evidence/batch-9-1-2/road_before_site.png'});
console.log('to site', await goto(311,190,3), await st());
await p.waitForTimeout(500);
await p.screenshot({path:'docs/qa/evidence/batch-9-1-2/ancient_site_entry.png'});
console.log('exit site', await goto(168,250), await st());
await p.waitForTimeout(500);
await p.screenshot({path:'docs/qa/evidence/batch-9-1-2/ancient_site_return_to_road.png'});
console.log('errors',logs);
await b.close();
